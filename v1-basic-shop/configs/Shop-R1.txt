! Shop-R1 - v1 basic shop configuration
! Router: Cisco 2911
! Network: 192.168.1.0/24 (static IPs, single LAN)

configure terminal
hostname Shop-R1

! Gateway for the shop LAN

 ip address 192.168.1.1 255.255.255.0
 no shutdown
end
